‍ #علائم_ظهور #آخر_الزمان #شبهات_مهدوی #قاتلین_سیدالشهداء #خط_فکری_یهود #رجعت #از_سقیفه_تا_کربلا ❓❕ آیا امام صادق علیه‌السلام از بعثت مردم ایران خبر داده‌اند؟ ┄┄┅═✧❁﷽❁✧═┅┄┄ متن روایت مورد استناد در کلیپ بالا : عَنْ أَبِي عَبْدِ اللَّهِ ع‌ فِي قَوْلِهِ تَعَالَى- وَ قَضَيْنا إِلى‌ بَنِي إِسْرائِيلَ‌ فِي الْكِتابِ‌ لَتُفْسِدُنَّ فِي الْأَرْضِ‌ مَرَّتَيْنِ‌قَالَ قَتْلُ عَلِيِّ بْنِ أَبِي طَالِبٍ ع وَ طَعْنُ الْحَسَنِ ع- وَ لَتَعْلُنَّ عُلُوًّا كَبِيراً قَالَ قَتْلُ الْحُسَيْنِ ع- فَإِذا جاءَ وَعْدُ أُولاهُما فَإِذَا جَاءَ نَصْرُ دَمِ الْحُسَيْنِ ع‌ بَعَثْنا عَلَيْكُمْ عِباداً لَنا أُولِي بَأْسٍ شَدِيدٍ فَجاسُوا خِلالَ الدِّيارِ قَوْمٌ يَبْعَثُهُمُ اللَّهُ قَبْلَ خُرُوجِ الْقَائِمِ ع فَلَا يَدَعُونَ وَتْراً لآِلِ مُحَمَّدٍ إِلَّا قَتَلُوهُ‌ وَ كانَ وَعْداً مَفْعُولًا خُرُوجُ الْقَائِمِ ع- ثُمَّ رَدَدْنا لَكُمُ الْكَرَّةَ عَلَيْهِمْ‌ خُرُوجُ الْحُسَيْنِ ع فِي سَبْعِينَ مِنْ أَصْحَابِهِ.... (تفسیر عیاشی، 2/ 281 - کافی، 8/ 206). ←مقدمتا آنکه در بسیاری از فضاهای سیاسی-دینی، نگاه آخرالزمانی به انواع رویدادها رواج دارد. این نگاه که در آن هر سخن مبهم یا نامرتبطی به طوری بازخوانی می‌شود که لحظۀ حاضر را توضیح دهد، ←می‌تواند در فهم واقعی مسائل و رویدادها ایجاد خطا کند. در اینجا به طور مشخص به یکی از نقل‌هایی توجه می‌کنیم که پیشتر هم ، در زمان جنگ ۱۲روزه در فضای مجازی نشر داده شد و اخیراً به عنوان پیش‌گویی امام صادق علیه‌السلام از «مبعوث‌شدن مردم ایران» (طبیعتاً در جنگ علیه آمریکا و اسرائیل) مطرح شده است. - این روایت نورانی از لسان معصوم علیه السلام از یک راوی متقدّم کوفی به نام «صالح بن سهل» در تفسیر آیه {وَقَضَيْنَا إِلَىٰ بَنِي إِسْرَائِيلَ فِي الْكِتَابِ لَتُفْسِدُنَّ فِي الْأَرْضِ مَرَّتَيْنِ وَلَتَعْلُنَّ عُلُوًّا كَبِيرًا} بیان شده است ؛ ✍🏻 در نگاه کلی در این حدیث شریف ، امام‌ صادق علیه‌السلام ذیل آیه ، اشاره به وقوع فساد در زمین به جهت قتل حضرت امیر و امام مجتبی علیهماالسلام میکنند و سپس اشاره به فساد بزرگتر ، شهادت سیدالشهداء سلام الله علیه می‌فرمایند و آنگاه خبر از خونخواهی از دشمنان اهل بیت علیهم‌السلام در آستانهٔ ظهور و سپس بحث رجعت سیدالشهداء و.... ⚠️ کسانی که معنای دلخواه خود را از حدیث فوق نشر داده و بدان استناد می‌کنند، باید پاسخ دهند که چگونه امام صادق علیه السلام قتل حضرت امیر علیه‌السّلام و حسنین علیهماالسّلام را به بنی اسرائیل نسبت داده اند؟! مگر جهودان امامان ما را به قتل رسانده اند؛ که قبل از ظهور حضرت مهدی علیه السلام، جهودان باید به انتقام آن خونها کشته شوند؟!! احادیث فراوانی از اهل بیت علیهم‌السلام پیرامون واقعهٔ کربلا صادر شده و در منابع متنوّع عامه و امامیه و زیدیه باقی مانده است. در بسیاری از این احادیث با توجه به شرایط امام و مخاطب‌شان، به صراحت انواع مظالم بنی‌امیه و ابن زیاد و دیگر قاتلان برشمرده شده و #تقیه در این خصوص کنار گذاشته شده است. در این میان وجه روشنی برای تقیه از یهود نیست و اگر یهود نقش فعالی در کربلا می‌داشت، انتظار می‌رفت که در روایاتی به این نکته تصریح شود. در مقابل از امام سجاد علیه‌السلام نقل شده که قاتلان امام حسین علیه السلام خود را از امت اسلام می‌پنداشتند و با ریختن خون امام حسین علیه السلام به خدا تقرّب می‌جستند (أمالی الصدوق، ص۵۴۷) و می‌دانیم که بسیاری از قاتلان آن حضرت دیدگاه‌های عثمانی داشته‌اند ؛ ➖اینکه ذیل آیه مرتبط با بنی اسرائیل این کلام بیان شده «تأویل آیه» است ، و یک جهت بیان آن ، این است که پیامبر خدا صلی‌الله علیه و آله وعده فرموده بودند که شما پا بر جای پای بنی اسرائیل می‌گذارید و سنت بنی اسرائیل را تکرار می‌کنید: 🔴 وَ فِی رِوَایَهًِْ سُلَیْمِ‌بْنِ‌قَیْسٍ الْهِلَالِیِّ عَنْ سَلْمَانَ الْفَارِسِیِّ أَنَّهُ قَالَ: ... ثُمَّ أَقْبَلَ عَلَیَّ سَلْمَانُ فَقَالَ ... سَمِعْتُ رَسُولَ اللَّهِ صلی الله علیه و آله یَقُولُ لَتَرْکَبَنَّ أُمَّتِی سُنَّهًَْ بَنِی إِسْرَائِیلَ حَذْوَ النَّعْلِ بِالنَّعْلِ {وَ حَذْوَ} الْقُذَّهًِْ بِالْقُذَّهًِْ شِبْراً بِشِبْرٍ وَ ذِرَاعاً بِذِرَاعِ وَ بَاعاً بِبَاعٍ. « #سلمان گوید: از پیامبر صلی الله علیه و آله شنیدم که می‌فرمودند : امّت من سنّت بنی‌اسرائیل را مرتکب خواهند شد به‌ طوری که قدم جای قدم آنان می‌ گذارند و تیر به همان‌جا که آنان زدند می‌ زنند ، و وجب به وجب و ذراع به ذراع و باع به باع کار های آنان را انجام خواهند داد.» 📔کتاب سلیم ۵۹۹/۲ 🔻در واقع این روایت بر نوعی تشبیه معنوی تأکید دارد. فسادهای بنی‌اسرائیل در طول تاریخ به معنای تجاوز از حدود الهی و ظلم در زمین بوده. ◄ ۱/۲ ►
